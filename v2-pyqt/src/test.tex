\section{Continuum SF}

We consider QCD defined on a manifold with boundary conditions
given by
\begin{equation}
\begin{gathered}
P_+ \psi(x) = \psibar (x) P_- = 0 \,, \quad x_0=0\\
P_- \psi(x) = \psibar (x) P_+ = 0 \,, \quad x_0=T
\end{gathered}
\end{equation}
The quark actions is given by
\begin{equation}
S = \int_0^T d x_0 \int d^3 {\mib x} \ 
\psibar (x) D \psi(x) \,, \quad 
D = \gamma_\mu (\partial_\mu + A_\mu) \,.
\end{equation}
and we consider here the homogeneous boundary conditions 
\footnote{With homogeneous BC we specify the 
non-zero components of the spinor at the boundaries. 
With inhomogeneous BC we introduce boundary fields 
like $P_+ \psi(x) =\rho(x) |_{x_0=0}$, compute 
expectation values of observables and then
set $\rho=0$.}
\begin{equation}
\begin{gathered}
\zeta(x) = P_- \psi(x) \,, \quad \zetabar(x) = \psibar(x) P_+ \,, \quad x_0=0 \\ 
\zeta^\prime(x) = P_+ \psi(x) \,, \quad \zetabar^\prime(x) = \psibar(x) P_- \,, \quad x_0=T 
\end{gathered}
\end{equation}
If we perform a non-singlet global chiral transformation 
\begin{equation}
\delta \psi(x) = \gamma_5 \lambda^a \psi(x) \,, \quad
\delta \psibar(x) = \psibar(x) \gamma_5 \lambda^a \,,
\end{equation}
assuming a theory with more than one flavor, 
the variation of the action reads
\begin{equation}
\delta S = - 
\int_0^T d x_0 \int d^3 {\mib x} \ 
\partial_\mu A_\mu^a(x_0,x) \,, \quad A^a_\mu = \psibar \gamma_\mu \gamma_5 \lambda^a \psi \,.
\end{equation}
The expectation value of a generic operator $\mathcal O$ 
under this transformation becomes
\begin{equation}
\langle \delta \mathcal O \rangle = 
\int_0^T d x_0 \int d^3 {\mib x} \ 
\langle \partial_\mu A_\mu^a(x) \mathcal O \rangle =  \int d^3 {\mib x} \
\langle \big[ A_0^a(x)|_{x_0=0} - A_0^a(x)|_{x_0=T} \big] O \rangle
\end{equation}
where we have used the fact the integral of the divergence yields only 
a surface term along the temporal direction due to the boundary conditions.
Due to the boundary conditions, such terms do not vanish, thus
completely breaking axial symmetries:
\begin{equation}
\begin{split}
A^a_0(x) \big|_{x_0=0} = & \psibar(x) P_+ \gamma_0 \gamma_5 \lambda^a P_- \psi(x) = 
+ \psibar(x) P_+ \gamma_5 \lambda^a P_- \psi(x) = 
+ \zetabar(x) \gamma_5 \lambda^a \zeta(x) |_{x_0=0} \\
A^a_0(x) \big|_{x_0=T} = &\psibar(x) P_- \gamma_0 \gamma_5 \lambda^a P_+ \psi(x) = 
- \psibar(x) P_- \gamma_5 \lambda^a P_+ \psi(x) = 
- \zetabar^\prime(x) \gamma_5 \lambda^a \zeta^\prime(x) |_{x_0=T}
\end{split}
\end{equation}
If we now choose $\mathcal O = \psi(x) \psibar(y)$ we obtain the following 
Ward-Identity
\begin{equation}
\begin{split}
\langle \delta O \rangle = \langle \gamma_5 \lambda^a \psi(x) \psibar(y) 
& + \psi(x) \psibar(y) \gamma_5 \lambda^a \rangle = \\ &  
\int d^3 {\mib z} \langle \psi(x) \psibar(y) \big\{
\zetabar(z) \gamma_5 \lambda^a \zeta(z) |_{z_0=0} + 
\zetabar^\prime(z) \gamma_5 \lambda^a \zeta^\prime(z) |_{z_0=T}
\big\} \rangle 
\end{split}
\end{equation}
From Wick's theorem we obtain the relevant fermion contractions, 
leading to a specific relation for the quark propagator $S$
\begin{equation}
\lambda^a \gamma_5 S(x,y) + S(x,y) \gamma_5 \lambda^a = 
\int d^3 {\mib z} S(x,z) \gamma_5 \lambda^a S(z,y) |_{z_0=0} + 
S(x,z) \gamma_5 \lambda^a S(z,y) |_{z_0=T}
\end{equation}
We now introduce the projector $P$, defined as
\begin{equation}
P f(x) = \int_0^T dx_0 \int d^3 {\mib x} 
\big\{P_- \delta(x_0-a) + P_+ \delta(x_0-T) \big\} f(x)
\end{equation}
and re-write the previous equation as follows
\begin{equation}
\lambda^a \gamma_5 S(x,y) + S(x,y) \gamma_5 \lambda^a = 
S(x,z) \gamma_5 \lambda^a P S(z,y) 
\end{equation}
Alternatively we also introduce a different projector 
\begin{equation}
\tilde P f(x) = \int_0^T dx_0 \int d^3 {\mib x}
\big\{\delta(x_0-a) + \delta(x_0-T) \big\} f(x)
\end{equation}
The two projectors are equivalent in our previous 
relation for the quark propagator, simply because
the quark propagator satisfies by definition the 
boundary conditions imposed on quark fields and the
projector $P$ simply corresponds to the non-zero 
components, namely
\begin{equation}
P_+ S(x,y) |_{x_0=0} = 0 \quad \to \quad 
S(x,y) |_{x_0=0} = P_- S(,xy) |_{x_0=0}
\end{equation}

\section{Overlap}

Now we introduce the mass-less overlap operator
\begin{equation}
a \Dov = \frac{1}{2} \big[ 1 + A (A^\dagger A)^{-1/2} \big]  \,, 
\end{equation}
whose kernel is defined from the Wilson-Dirac operator $D_W$ with a negative
mass  term
\begin{equation}
A = a D_W - a M \,, \quad 0<aM<2 \,.
\end{equation}
In infinite volume the overlap operator satisfies the Ginsparg-Wilson 
relation exactly
\begin{equation}
\Dov \gamma_5 + \gamma_5 \Dov = a \Dov \gamma_5 \Dov
\end{equation}

The massive overlap operator can be defined as
\begin{equation}
a \Dov(m) = \frac{1}{2} \Big[ (1 + a m) + (1-a m) A (A^\dagger A)^{-1/2} \Big]
\end{equation}
which can be re-written as follows
\begin{equation}
\Dov(m) = \Dov (1- am) + m = \mu \Dov + m
\end{equation}

How is the Ginsparg-Wilson modified by the mass term?
\begin{equation}
\begin{gathered}
\gamma_5 \Dov(m) + \Dov(m) \gamma_5 - a \Dov(m) \gamma_5 \Dov(m) = \\
\mu ( \gamma_5 \Dov + \Dov \gamma_5 ) + 2 m \gamma_5 - \mu^2 a \Dov \gamma_5 \Dov
- am \mu (\Dov \gamma_5 + \gamma_5 \Dov ) - am^2 \gamma_5 = \\  
a \Dov \gamma_5 \Dov (\mu - \mu^2 -a m \mu ) + m(2 - am) \gamma_5 = \\
a \Dov \gamma_5 \Dov [(1-am) \mu - \mu^2] + m(2 - am) \gamma_5
\end{gathered}
\end{equation}
The massive overlap operator obeys the modified ginsparg-Wilson 
relation
\begin{equation}
\gamma_5 a \Dov(m) + a \Dov(m) \gamma_5 = a^2 \Dov(m) \gamma_5 \Dov(m) + c \gamma_5
\end{equation}
where $c$ is a dimension-less constant given by $c=am (2-am)$.

If we now assume that the underlying operator $D_W$ corresponds to 
the SF operator, i.e. defined only in the interior of the 
lattice at $a \leq x_0 \leq T-2a$, the standard definition of the mass-less
overlap operator leads, by definition, to an operator that satisfies 
the Ginsparg-Wilson relation.
Here we propose to modify the overlap definition with a mass term 
localized at the boundaries, namely
\begin{equation}
\Dov^\mathrm{SF} = \frac{1}{2} [(1 + c m(x_0)) + (1-c m(x_0)) A (A^\dagger A)^{-1/2} ] \,, 
\quad m(x_0) = \left\{ \begin{array}{cl} 1 & x_0=a \\
0 & a<x_0<T-a \\
1 & x_0=T-a\\
\end{array} \right.
\end{equation}
with $c$ a tunable coefficient.
Such an operator obeys the following modified Ginsparg-Wilson relation
\begin{equation}
\gamma_5 \Dovsf + \Dovsf \gamma_5 = a \Dovsf \gamma_5 \Dovsf + a \tilde c \gamma_5 P \,,
\end{equation}
\begin{equation}
P = \frac{1}{a} [\delta_{x_0,a} + \delta_{x_0,T-a}]
\end{equation}

We can immediately check that the propagator defined from $\Dovsf$ 
obeys the following relation
\begin{equation}
S(x,y) \gamma_5  + \gamma_5 S(x,y) = \gamma_5 \delta_{x,y} + 
\tilde c \big[ S (x,z) \gamma_5 S(z,y)|_{z_0=0}  
+ S (x,z) \gamma_5 S(z,y)|_{z_0=T} \big]
\end{equation}
